/* ========================================================================== ROBUST EXCEL IMPORT & EXPORT (SheetJS) ========================================================================== */ // Helper to normalize month names (handles "Aug 2025", "Aug-25", "August 2025", Excel serial numbers) function normalizeMonthHeader(rawVal) { if (rawVal === undefined || rawVal === null) return null; // 1. If Excel stored it as a numeric serial date if (typeof rawVal === "number" && rawVal > 30000 && rawVal < 70000) { const jsDate = new Date(Math.round((rawVal - 25569) * 86400 * 1000)); const mName = MONTH_NAMES[jsDate.getUTCMonth()]; const yNum = jsDate.getUTCFullYear(); return `${mName} ${yNum}`; } const str = String(rawVal).trim(); if (!str) return null; // 2. Direct match with MONTH_LIST (e.g., "Aug 2025") for (const m of MONTH_LIST) { if (m.toLowerCase() === str.toLowerCase()) return m; } // 3. Regex match for "Aug-25", "August 2025", "08/2025", etc. const monthRegex = /(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*[\s\-_',./]*(\d{2,4})/i; const match = str.match(monthRegex); if (match) { const monShort = match[1].charAt(0).toUpperCase() + match[1].slice(1, 3).toLowerCase(); let year = parseInt(match[2], 10); if (year < 100) year += 2000; // Convert 25 to 2025 const candidate = `${monShort} ${year}`; if (MONTH_LIST.includes(candidate)) return candidate; } return null; } // Helper to parse numeric money values (handles "Rs. 2,000", "2000", blanks) function parseCleanAmount(val) { if (val === undefined || val === null) return 0; if (typeof val === "number") return isNaN(val) ? 0 : val; const cleaned = String(val).replace(/[^0-9.-]/g, ""); const num = parseFloat(cleaned); return isNaN(num) ? 0 : num; } /* --- EXPORT TO EXCEL --- */ function exportToExcel() { try { const wb = XLSX.utils.book_new(); // 1. Build "Monthly Entry" Sheet (Col A = Sr #, Col B = Member Name, Col C+ = Months) const monthlyAOA = []; const headerRow = ["Sr #", "Member Name", ...MONTH_LIST]; monthlyAOA.push(headerRow); appData.members.forEach((member, index) => { const row = [index + 1, member]; MONTH_LIST.forEach(month => { const val = (appData.deposits[member] && appData.deposits[member][month]) || ""; row.push(val); }); monthlyAOA.push(row); }); const wsMonthly = XLSX.utils.aoa_to_sheet(monthlyAOA); XLSX.utils.book_append_sheet(wb, wsMonthly, "Monthly Entry"); // 2. Build "Expenditure" Sheet (Col A = Sr #, Col B = Item Description, Col C = Price) const expAOA = [["Sr #", "Item Description", "Price"]]; appData.expenses.forEach((e, idx) => { expAOA.push([idx + 1, e.name, e.price]); }); const wsExp = XLSX.utils.aoa_to_sheet(expAOA); XLSX.utils.book_append_sheet(wb, wsExp, "Expenditure"); // 3. Save file XLSX.writeFile(wb, "Family_Savings_Fund_Backup.xlsx"); } catch (err) { alert("Export failed: " + err.message); console.error("Export Error:", err); } } /* --- IMPORT FROM EXCEL --- */ function importFromExcel(event) { const file = event.target.files && event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onerror = function() { alert("File Reading Error: Unable to read the selected file from disk."); }; reader.onload = function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: "array", cellDates: true }); if (!workbook.SheetNames || workbook.SheetNames.length === 0) { throw new Error("The selected Excel workbook has no sheets."); } // Locate sheets flexibly (case-insensitive search) const findSheet = (patterns) => { for (const name of workbook.SheetNames) { const lower = name.trim().toLowerCase(); for (const p of patterns) { if (lower.includes(p.toLowerCase())) return workbook.Sheets[name]; } } return null; }; const sheetMonthly = findSheet(["Monthly Entry", "MonthlyEntry", "Monthly", "Entry"]) || workbook.Sheets[workbook.SheetNames[0]]; const sheetExpenditure = findSheet(["Expenditure", "Expenditures", "Expense", "Expenses"]); // ========================================================== // PARSE SHEET 1: MONTHLY ENTRY // ========================================================== if (!sheetMonthly) { throw new Error("Could not find the 'Monthly Entry' sheet in this workbook."); } // Read raw grid const rawMonthlyGrid = XLSX.utils.sheet_to_json(sheetMonthly, { header: 1, defval: "" }); if (!rawMonthlyGrid || rawMonthlyGrid.length < 2) { throw new Error("'Monthly Entry' sheet is empty or missing data rows."); } // Step A: Find the Header Row containing months and identify the Member Name column let headerRowIndex = -1; let memberColIndex = 0; // default Col A const monthColMap = {}; // colIndex -> "Aug 2025" for (let r = 0; r < Math.min(rawMonthlyGrid.length, 10); r++) { const row = rawMonthlyGrid[r]; let detectedMonths = 0; for (let c = 0; c < row.length; c++) { const norm = normalizeMonthHeader(row[c]); if (norm) detectedMonths++; } // If row contains multiple month headers, this is our header row if (detectedMonths >= 2) { headerRowIndex = r; break; } } if (headerRowIndex === -1) { throw new Error("Could not detect Month columns (e.g. 'Aug 2025', 'Sep 2025') in the top rows of 'Monthly Entry'."); } const headerRow = rawMonthlyGrid[headerRowIndex]; // Step B: Determine if Member Name is in Col A (0) or Col B (1) // Check column header text or data characteristics let col0Header = String(headerRow[0] || "").toLowerCase(); let col1Header = String(headerRow[1] || "").toLowerCase(); if (col1Header.includes("member") || col1Header.includes("name")) { memberColIndex = 1; } else if (col0Header.includes("member") || col0Header.includes("name")) { memberColIndex = 0; } else { // Inspect row immediately below header: if Col 0 is a number (1, 2) and Col 1 is text, names are in Col B const sampleRow = rawMonthlyGrid[headerRowIndex + 1]; if (sampleRow && !isNaN(sampleRow[0]) && String(sampleRow[0]).trim() !== "" && isNaN(sampleRow[1])) { memberColIndex = 1; } else { memberColIndex = 0; } } // Map month columns for (let c = 0; c < headerRow.length; c++) { if (c === memberColIndex) continue; const norm = normalizeMonthHeader(headerRow[c]); if (norm) { monthColMap[c] = norm; } } // Step C: Extract Members and Deposits const parsedMembers = []; const parsedDeposits = {}; const skipKeywords = ["total", "grand total", "subtotal", "sum", "average", "balance"]; for (let r = headerRowIndex + 1; r < rawMonthlyGrid.length; r++) { const row = rawMonthlyGrid[r]; if (!row || row.length === 0) continue; let memberName = String(row[memberColIndex] || "").trim(); // Fallback: If Col B was chosen but empty, try Col A if (!memberName && memberColIndex === 1 && isNaN(row[0])) { memberName = String(row[0] || "").trim(); } if (!memberName) continue; if (skipKeywords.includes(memberName.toLowerCase())) continue; if (!parsedMembers.includes(memberName)) { parsedMembers.push(memberName); parsedDeposits[memberName] = {}; } // Read deposits for this member Object.keys(monthColMap).forEach(colIdx => { const monthName = monthColMap[colIdx]; const amount = parseCleanAmount(row[colIdx]); if (amount > 0) { parsedDeposits[memberName][monthName] = amount; } }); } if (parsedMembers.length === 0) { throw new Error("No member names could be found. Please check whether Member Names are in Column A or Column B."); } // ========================================================== // PARSE SHEET 2: EXPENDITURE (Optional / Auto-detected) // ========================================================== const parsedExpenses = []; if (sheetExpenditure) { const rawExpGrid = XLSX.utils.sheet_to_json(sheetExpenditure, { header: 1, defval: "" }); for (let r = 0; r < rawExpGrid.length; r++) { const row = rawExpGrid[r]; if (!row || row.length < 2) continue; // Find first text column that isn't a header or total let itemName = ""; let itemPrice = 0; // Check if Col A is Sr and Col B is Name, or Col A is Name const str0 = String(row[0] || "").trim(); const str1 = String(row[1] || "").trim(); const num2 = parseCleanAmount(row[2]); if (str0 && skipKeywords.includes(str0.toLowerCase())) continue; if (str1 && skipKeywords.includes(str1.toLowerCase())) continue; if (str1 && num2 > 0) { // [Sr, Item, Price] itemName = str1; itemPrice = num2; } else { // [Item, Price] const num1 = parseCleanAmount(row[1]); if (str0 && num1 > 0 && isNaN(str0)) { itemName = str0; itemPrice = num1; } } // Skip table headers if (itemName.toLowerCase().includes("item") || itemName.toLowerCase().includes("description")) continue; if (itemName && itemPrice > 0) { parsedExpenses.push({ id: "e" + (parsedExpenses.length + 1) + "_" + Date.now(), name: itemName, price: itemPrice }); } } } // ========================================================== // COMMIT TO APPLICATION STATE // ========================================================== appData.members = parsedMembers; appData.deposits = parsedDeposits; if (parsedExpenses.length > 0) { appData.expenses = parsedExpenses; } saveState(); refreshAllScreens(); const metrics = getFundMetrics(); alert( `Import Successful!\n\n` + `• Active Members: ${metrics.membersCount}\n` + `• Total Collected: ${formatRs(metrics.totalCollected)}\n` + `• Total Spent: ${formatRs(metrics.totalSpent)}\n` + `• Available Balance: ${formatRs(metrics.availableBalance)}\n` + `• Latest Month: ${metrics.latestMonth} (${metrics.latestPaidCount} paid)` ); } catch (err) { console.error("Detailed Excel Import Error:", err); alert( "Import Error:\n" + err.message + "\n\n" + "Tip: Check the browser console (F12 or inspect) for the complete log." ); } finally { // Clear file input so the same file can be selected again if needed event.target.value = ""; } }; reader.readAsArrayBuffer(file); }
